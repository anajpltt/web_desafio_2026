esc <- 10

2. Vetores

# Criar um vetor numérico
num <- c(10, 2.5, 4, esc)
num2 <- c(3,4,2,2)

typeof(num)
[1] "double"
#double = variáveis quantitativas contínuas

# Operação com vetor e escalar
# ex. transformar a unidade de variáveis
num + esc
[1] 20.0 12.5 14.0 20.0
num / esc
[1] 1.00 0.25 0.40 1.00
num - esc
[1]  0.0 -7.5 -6.0  0.0
num ^ esc
[1] 1.000000e+10 9.536743e+03 1.048576e+06 1.000000e+10
# Operação pairwise/element-wise
num + num2
[1] 13.0  6.5  6.0 12.0
num - num2
[1]  7.0 -1.5  2.0  8.0
num / num2
[1] 3.333333 0.625000 2.000000 5.000000
num * num2
[1] 30 10  8 20
# Reciclagem (operação recilada)
##O menor vetor é utilizado 2x na operação
num3 <- c(num, num2)
num * num3
[1] 100.00   6.25  16.00 100.00  30.00  10.00   8.00  20.00

3. Caracteres/fatores lógicos

char <- c("t1, t2, t3")
typeof(char)
[1] "character"
char2 <- c("pequeno", "médio", "grande", "gigante")
sort(char2) #ordenar vetor texto em ordem alfabética
[1] "gigante" "grande"  "médio"   "pequeno"
# Fator
as.factor(char2) #transformar em fator
[1] pequeno médio   grande  gigante
Levels: gigante grande médio pequeno
factor(char2, levels = c("pequeno", "médio", "grande", "gigante")) #transformar em fatores e definir a ordem
[1] pequeno médio   grande  gigante
Levels: pequeno médio grande gigante
# fator numérico
factor(num)
[1] 10  2.5 4   10 
Levels: 2.5 4 10

2. Matrizes

# Matrizes são obrigatoriamente numéricas

# Matriz com variável normal (média 30, dp 4)
set.seed(1)
mat <- matrix(data = rnorm(n = 16, mean = 30, sd = 4),
       nrow = 4,
       ncol = 4)
#os dados são para exemplificar, por isso rnorm está gerando números aleatórios para preencher a matriz
#set.seed torna a aleatoriedade dos números reprodutível

# Seleção de elementos

alunos[4,2] #[linha, coluna]
[1] 160
# Seleção de linhas
alunos[3,]

3. Listas

lista <-
  list(
  dataframe = alunos,
  matriz = mat,
  escalar = esc,
  numeric = num,
  fator = factor(char),
  lista = list(df = alunos, matriz = mat))

#lista = "armário'" que permite guardar diferentes tipos de objetos em diferentes gavetas

# catar diferentes coisas de dentro da lista usando $
lista$dataframe
      nome altura cromossomo
1   thiago    191         xy
2 gabriela    162         xx
3  jamille    169         xx
4      ana    160         xx
5  richard    190         xy
6   cleibi    188         xy
# criar um armário novo com uma única gaveta
ex1 <- lista[6]

# tirar o objeto da gaveta que está dentro do armário (nesse caso, o dataframe)
ex2 <- lista[[6]][[1]]

# simulação de um excel com várias abas (lista com vários df)
dados <- 
  list(
  aba1 = alunos,
  aba2 = alunos,
  aba3 = alunos
  )

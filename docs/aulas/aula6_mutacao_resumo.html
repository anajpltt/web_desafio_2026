# espacamento de 0.2 entre linhas
# 50 plantas por metro linear
# calcular numero de plantas por ha (npha)
# calcular rendimento em sacas (60kg) por ha (rgsha)
dfrg <-
df |>
  mutate(gen2 = str_replace(gen, "G", "GEN_" ),
         .after = gen
         ) |>
  mutate(cuidado = case_when(mc > mg ~ "Não", .default = "Sim"),
         icc = mg / mc,
         npha = 100/0.2*100*50,
         rgsha = mg * npha / 1000 / 60)
  
quant <- quantile(dfrg$rgsha, c(0.05, 0.25, 0.5, 0.75, 0.95), na.rm = TRUE)

dfrg <-
  dfrg |>
  mutate(
  classe = cut(dfrg$rgsha,
               breaks = c(-Inf, quant, Inf),
               labels = c("Muito ruim", "Ruim", "Médio", "Bom", "Muito bom", "ótimo"),
               include.lowest = TRUE)  
  )

cv <- function(x, na.rm = TRUE){
  sd(x, na.rm = na.rm) / mean(x, na.rm = na.rm) * 100
  }

confint <- function(x, na.rm = TRUE, return = "inf") {
  n <- length(na.omit(x))
  ttab <- qt(0.975, n - 1)
  epm <- sd(x, na.rm = na.rm) / sqrt(n)
  media <- mean(x, na.rm = na.rm)
  if(return == "inf") {
    media - ttab * epm }
  else{
    media + ttab * epm
  }
}

Usando across

stats <- 
  list(
    media = \(x){mean(x, na.rm = TRUE)},
    min = \(x){min(x, na.rm = TRUE)},
    max = \(x){max(x, na.rm = TRUE)},
    dp = \(x){sd(x, na.rm = TRUE)},
    var = \(x){var(x, na.rm = TRUE)},
    cv = \(x){cv(x, na.rm = TRUE)},
    li = \(x){confint(x)},
    ls= \(x){confint(x, return = "sup")},
    n = \(x){length(x)},
    nvalid = \(x){length(na.omit(x))})


dfrg |> 
  summarise(
    across(mg, .fns = stats)
